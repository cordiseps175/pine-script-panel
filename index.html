15m ve Altı Zaman Dilimleri • EMA/RSI Erken Dönüş • Hedef Çizgisi/Etiketi
//@version=6
strategy("15m Altı Erken EMA/RSI Dönüş + Hedef Etiketi", overlay=true, max_lines_count=500, max_labels_count=500)

// Zaman dilimi kontrolü (15 dakika ve altı)
isAllowedTf = timeframe.isintraday and timeframe.multiplier <= 15

// Girdiler
fastLen     = input.int(9, "Hızlı EMA", minval=1)
slowLen     = input.int(21, "Yavaş EMA", minval=1)
rsiLen      = input.int(7, "Erken RSI Periyodu", minval=2)
rsiOversold = input.float(32.0, "RSI Aşırı Satım Eşiği", step=0.5)
rsiOverbought = input.float(68.0, "RSI Aşırı Alım Eşiği", step=0.5)
rrTarget    = input.float(2.0, "Hedef R:R Oranı", step=0.25, minval=0.5)
targetBars  = input.int(12, "Hedef Bitiş Çizgisi Mum Uzunluğu", minval=2, maxval=100)

// Göstergeler
emaFast = ta.ema(close, fastLen)
emaSlow = ta.ema(close, slowLen)
rsiVal  = ta.rsi(close, rsiLen)

// Erken dönüş koşulları
bullEarlyTurn = isAllowedTf and (ta.crossover(rsiVal, rsiOversold) or (close > emaFast and emaFast > emaSlow and rsiVal > rsiVal[1]))
bearEarlyTurn = isAllowedTf and (ta.crossunder(rsiVal, rsiOverbought) or (close < emaFast and emaFast < emaSlow and rsiVal < rsiVal[1]))

// Giriş ve stop/hedef hesaplama
atrVal = ta.atr(14)

if bullEarlyTurn and strategy.position_size <= 0
    stopDist = math.max(atrVal * 1.2, syminfo.mintick * 10)
    slPrice  = close - stopDist
    tpPrice  = close + (stopDist * rrTarget)
    strategy.entry("Long", strategy.long)
    strategy.exit("Exit Long", from_entry="Long", stop=slPrice, limit=tpPrice)

    endBar = bar_index + targetBars
    line.new(bar_index, tpPrice, endBar, tpPrice, xloc=xloc.bar_index, extend=extend.none, color=color.new(color.lime, 0), style=line.style_dashed, width=2)
    label.new(endBar, tpPrice, "TP Long: " + str.tostring(tpPrice, format.mintick), xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left, color=color.new(color.lime, 15), textcolor=color.black, size=size.small)

if bearEarlyTurn and strategy.position_size >= 0
    stopDist = math.max(atrVal * 1.2, syminfo.mintick * 10)
    slPrice  = close + stopDist
    tpPrice  = close - (stopDist * rrTarget)
    strategy.entry("Short", strategy.short)
    strategy.exit("Exit Short", from_entry="Short", stop=slPrice, limit=tpPrice)

    endBar = bar_index + targetBars
    line.new(bar_index, tpPrice, endBar, tpPrice, xloc=xloc.bar_index, extend=extend.none, color=color.new(color.red, 0), style=line.style_dashed, width=2)
    label.new(endBar, tpPrice, "TP Short: " + str.tostring(tpPrice, format.mintick), xloc=xloc.bar_index, yloc=yloc.price, style=label.style_label_left, color=color.new(color.red, 15), textcolor=color.white, size=size.small)

// EMA çizimleri
plot(emaFast, "Hızlı EMA", color=color.new(#38bdf8, 0), linewidth=1)
plot(emaSlow, "Yavaş EMA", color=color.new(#f59e0b, 0), linewidth=2)

// 15m üstü uyarı etiketi
var label tfWarn = na
if not isAllowedTf and barstate.islast
    label.delete(tfWarn)
    tfWarn := label.new(bar_index, high, "Bu strateji 15m ve altı zaman dilimleri içindir", style=label.style_label_down, color=color.new(color.orange, 10), textcolor=color.black, size=size.small)
